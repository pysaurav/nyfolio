With it, a post at worst emits its own opening sentence. */}} {{/* Each candidate is normalised BEFORE it is tested for emptiness. -}}{{- $desc = $norm -}}{{- end -}} {{- end -}} {{- end -}} {{- if and (not $desc) $p.IsPage -}} {{- with $p.Summary -}} {{- $norm = trim (. | plainify | htmlUnescape | replaceRE `\s+` " ") " " -}} than by discipline. */}} {{- $src := "" -}} {{- with $p.Params.images -}}{{- $src = index . 0 -}}{{- end -}} {{- if not $src -}}{{- with $p.Params.image -}}{{- $src = . -}}{{- end -}}{{- end -}} {{- if not $src -}} {{- with $p.Resources.GetMatch "{cover,og,feature}.{png,jpg,jpeg,webp}" -}} {{- $src = .RelPermalink -}} {{- end -}} {{- end -}} {{/* Posts fall back to a distinct card so an article $r.Width -}}{{- $ogH = $r.Height -}} {{- $ogType = $r.MediaType.Type -}} {{- else -}} {{/* static/ or content/ path: passthrough, no processing available */}} {{- $ogURL = $src | absURL -}} {{- if or
